#!/usr/bin/env bash

# cap add ios esegue pod install internamente come ultimo step.
# Se il deployment target è troppo basso, pod install fallisce ma il
# progetto Xcode + Podfile vengono comunque creati → ignoriamo l'errore.
if [ ! -d "ios" ]; then
  npx cap add ios || true
fi
# Verifica che il Podfile esista prima di modificarlo
if [ ! -f "ios/App/Podfile" ]; then
  echo "ERRORE: Podfile non trovato dopo cap add ios"
  exit 1
fi
# CapacitorStatusBar e altri plugin richiedono iOS 14+
sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '14.0'/" ios/App/Podfile
echo "Podfile aggiornato:"
grep "platform :ios" ios/App/Podfile
# Ora eseguiamo pod install con il target corretto
cd ios/App && pod install
